cmake_minimum_required(VERSION 3.22)
project(feature_protection_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(MESHOPTIMIZER_INCLUDE_DIR "${REPO}/build/_deps/nvpro_core2/third_party/meshoptimizer/src" CACHE PATH "meshoptimizer headers")
set(MESHOPTIMIZER_LIBRARY "${REPO}/build/nvpro_core2/third_party/meshoptimizer/Release/meshoptimizer.lib" CACHE FILEPATH "meshoptimizer library")
add_executable(feature_invariant_tests invariant_tests.cpp)
target_include_directories(feature_invariant_tests PRIVATE "${REPO}/src/meshlod" "${MESHOPTIMIZER_INCLUDE_DIR}")
target_link_libraries(feature_invariant_tests PRIVATE "${MESHOPTIMIZER_LIBRARY}")
target_compile_definitions(feature_invariant_tests PRIVATE NOMINMAX)
enable_testing()
add_test(NAME feature_invariants COMMAND feature_invariant_tests)
